train_path = r"F:\my item\O2O优惠券个性化投放\【下载】项目数据文件\train.csv"
df = pd.read_csv(train_path)

print("=" * 60)
print("【数据读取完成】")
print(f"数据形状: {df.shape}")

print("\n>>> 数值型变量描述统计:")
print(df.describe().round(2))

missing = df.isnull().sum()

# 3. 日期转换与基础标识
# ============================================

def convert_date(x):
    """将数值日期转为 datetime"""
    if pd.isna(x) or str(x).lower() in ('nan', 'null', 'none'):
        return pd.NaT
    try:
        return pd.to_datetime(str(int(float(x))), format='%Y%m%d')
    except:
        return pd.NaT

df['date_dt'] = df['date'].apply(convert_date)
df['date_received_dt'] = df['date_received'].apply(convert_date)

df['consume_year'] = df['date_dt'].dt.year
df['consume_month'] = df['date_dt'].dt.month
df['receive_month'] = df['date_received_dt'].dt.month

# 基础标识
df['has_coupon'] = df['coupon_id'].notna() & df['date_received'].notna()
df['is_consumed'] = df['date'].notna()
df['is_coupon_consumed'] = df['has_coupon'] & df['is_consumed']

print("\n日期转换与标识完成:")
print(f"  有效消费记录: {df['is_consumed'].sum():,}")
print(f"  有效领券记录: {df['has_coupon'].sum():,}")
print(f"  领券后消费记录: {df['is_coupon_consumed'].sum():,}")

# 4. 图2：2016年用户每个月消费次数折线图
# ============================================

print("\n" + "=" * 60)
print("【2. 绘制2016年用户每月消费次数折线图】")
print("=" * 60)

df_2016 = df[df['consume_year'] == 2016]
monthly_consume = df_2016[df_2016['is_consumed']].groupby('consume_month').size().reset_index(name='consume_count')

fig, ax = plt.subplots(1, 1, figsize=(10, 6))

ax.plot(monthly_consume['consume_month'], monthly_consume['consume_count'],
        marker='o', markersize=8, color='#4ECDC4', linewidth=2.5)
ax.fill_between(monthly_consume['consume_month'], monthly_consume['consume_count'],
                alpha=0.3, color='#4ECDC4')

for _, row in monthly_consume.iterrows():
    ax.annotate(f"{int(row['consume_count']):,}",
                xy=(row['consume_month'], row['consume_count']),
                xytext=(0, 10), textcoords='offset points',
                ha='center', fontsize=9, fontweight='bold', color='#2C3E50')

ax.set_title('2016年用户每月消费次数趋势', fontsize=14, fontweight='bold', pad=15)
ax.set_xlabel('月份', fontsize=12)
ax.set_ylabel('消费次数', fontsize=12)
ax.set_xticks(range(1, 13))
ax.set_xticklabels([f'{i}月' for i in range(1, 13)])
ax.grid(True, alpha=0.3, linestyle='--')

plt.tight_layout(pad=3.0)
plt.savefig('chart2_2016_monthly_consume.png', dpi=200, bbox_inches='tight')
plt.show()

# 各月领券次数
monthly_receive = df[df['has_coupon']].groupby('receive_month').size().reset_index(name='receive_count')
# 各月领券消费次数
monthly_coupon_consume = df[df['is_coupon_consumed']].groupby('receive_month').size().reset_index(name='coupon_consume_count')
# 合并
monthly_coupon = pd.merge(monthly_receive, monthly_coupon_consume, on='receive_month', how='outer').fillna(0).sort_values('receive_month')

# 各月投放优惠券总数（即被领取的优惠券总数，按月份统计）
monthly_sent = df[df['has_coupon']].groupby('receive_month').size().reset_index(name='total_sent').sort_values('receive_month')

fig, axes = plt.subplots(1, 2, figsize=(16, 6))

# 左图：各月领券次数 vs 领券消费次数
ax1 = axes[0]
x = np.arange(len(monthly_coupon))
width = 0.35

bars1 = ax1.bar(x - width/2, monthly_coupon['receive_count'], width,
                label='领券次数', color='#FF6B6B', alpha=0.85, edgecolor='white')
bars2 = ax1.bar(x + width/2, monthly_coupon['coupon_consume_count'], width,
                label='领券消费次数', color='#4ECDC4', alpha=0.85, edgecolor='white')

for bars in [bars1, bars2]:
    for bar in bars:
        height = bar.get_height()
        ax1.annotate(f'{int(height):,}',
                     xy=(bar.get_x() + bar.get_width()/2, height),
                     xytext=(0, 3), textcoords='offset points',
                     ha='center', fontsize=8, fontweight='bold')

ax1.set_title('各月用户领券次数与领券消费次数', fontsize=14, fontweight='bold', pad=15)
ax1.set_xlabel('月份', fontsize=12)
ax1.set_ylabel('次数', fontsize=12)
ax1.set_xticks(x)
ax1.set_xticklabels([f"{int(m)}月" for m in monthly_coupon['receive_month']])
ax1.legend(fontsize=11)
ax1.grid(True, alpha=0.3, linestyle='--', axis='y')

# 右图：各月商家投放优惠券总数
ax2 = axes[1]
bars3 = ax2.bar(monthly_sent['receive_month'], monthly_sent['total_sent'],
                color='#45B7D1', alpha=0.85, edgecolor='white', linewidth=1)

for bar in bars3:
    height = bar.get_height()
    ax2.annotate(f'{int(height):,}',
                 xy=(bar.get_x() + bar.get_width()/2, height),
                 xytext=(0, 3), textcoords='offset points',
                 ha='center', fontsize=9, fontweight='bold')

ax2.set_title('各月商家投放优惠券总数', fontsize=14, fontweight='bold', pad=15)
ax2.set_xlabel('月份', fontsize=12)
ax2.set_ylabel('投放总数', fontsize=12)
ax2.set_xticks(monthly_sent['receive_month'])
ax2.set_xticklabels([f"{int(m)}月" for m in monthly_sent['receive_month']])
ax2.grid(True, alpha=0.3, linestyle='--', axis='y')

plt.tight_layout(pad=3.0)
plt.savefig('chart3_monthly_coupon.png', dpi=200, bbox_inches='tight')
plt.show()

# 6. 图4：距离比例饼图（总体 + 持券 + 未持券）
# ============================================

print("\n" + "=" * 60)
print("【4. 绘制用户到门店消费距离比例饼图】")
print("=" * 60)

def cat_dist(d):
    if pd.isna(d):
        return '未知'
    d = int(float(d))
    if d == 0:
        return '0km(同区域)'
    elif d <= 1:
        return '1km以内'
    elif d <= 3:
        return '1-3km'
    elif d <= 5:
        return '3-5km'
    elif d <= 10:
        return '5-10km'
    else:
        return '10km以上'

# 筛选有消费记录的数据
consume_df = df[df['is_consumed']].copy()
consume_df['dist_group'] = consume_df['distance'].apply(cat_dist)

# 分离持券 / 未持券
coupon_consume = consume_df[consume_df['has_coupon']].copy()
direct_consume = consume_df[~consume_df['has_coupon']].copy()

# 统计
all_dist = consume_df['dist_group'].value_counts()
cp_dist = coupon_consume['dist_group'].value_counts()
dc_dist = direct_consume['dist_group'].value_counts()

# 统一排序
order = ['0km(同区域)', '1km以内', '1-3km', '3-5km', '5-10km', '10km以上', '未知']
all_dist = all_dist.reindex(order, fill_value=0)
cp_dist = cp_dist.reindex(order, fill_value=0)
dc_dist = dc_dist.reindex(order, fill_value=0)

all_plot = all_dist[all_dist > 0]
cp_plot = cp_dist[cp_dist > 0]
dc_plot = dc_dist[dc_dist > 0]

colors = ['#FF6B6B', '#4ECDC4', '#45B7D1', '#96CEB4', '#FFD93D', '#FF9F43', '#A29BFE']

fig, axes = plt.subplots(1, 3, figsize=(20, 6))

# 左图：总体
ax1 = axes[0]
w1, t1, at1 = ax1.pie(
    all_plot.values,
    labels=all_plot.index,
    autopct='%1.1f%%',
    startangle=90,
    colors=colors[:len(all_plot)],
    shadow=True,
    explode=[0.05 if i == all_plot.values.argmax() else 0 for i in range(len(all_plot))],
    textprops={'fontsize': 10}
)
for at in at1:
    at.set_color('white')
    at.set_fontweight('bold')
    at.set_fontsize(9)
ax1.set_title('用户到门店消费距离比例\n(总体)', fontsize=13, fontweight='bold', pad=15)

# 中图：持券用户
ax2 = axes[1]
w2, t2, at2 = ax2.pie(
    cp_plot.values,
    labels=cp_plot.index,
    autopct='%1.1f%%',
    startangle=90,
    colors=colors[:len(cp_plot)],
    shadow=True,
    explode=[0.05 if i == cp_plot.values.argmax() else 0 for i in range(len(cp_plot))],
    textprops={'fontsize': 10}
)
for at in at2:
    at.set_color('white')
    at.set_fontweight('bold')
    at.set_fontsize(9)
ax2.set_title('持券用户到门店消费\n距离比例', fontsize=13, fontweight='bold', pad=15, color='#C0392B')

# 右图：未持券用户
ax3 = axes[2]
w3, t3, at3 = ax3.pie(
    dc_plot.values,
    labels=dc_plot.index,
    autopct='%1.1f%%',
    startangle=90,
    colors=colors[:len(dc_plot)],
    shadow=True,
    explode=[0.05 if i == dc_plot.values.argmax() else 0 for i in range(len(dc_plot))],
    textprops={'fontsize': 10}
)
for at in at3:
    at.set_color('white')
    at.set_fontweight('bold')
    at.set_fontsize(9)
ax3.set_title('未持券用户到门店消费\n距离比例', fontsize=13, fontweight='bold', pad=15, color='#27AE60')

plt.tight_layout(pad=3.0)
plt.savefig('chart4_distance_pie.png', dpi=200, bbox_inches='tight')
plt.show()

print(f"图表已保存: chart4_distance_pie.png")
